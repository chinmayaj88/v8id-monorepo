generator client {
  provider = "prisma-client"
  output   = "../generated/prisma"
}

datasource db {
  provider = "mysql"
}

// Enums
enum UserRole {
  USER
  ADMIN
}

enum DeviceType {
  MOBILE
  WEB
}

// User Model
model User {
  id                   String    @id @default(cuid())
  email                String    @unique
  passwordHash         String
  firstName            String?
  lastName             String?
  avatarUrl            String?
  role                 UserRole  @default(USER)
  emailVerified        Boolean   @default(true) // Admin creates verified users
  passwordResetToken   String?
  passwordResetExpires DateTime?
  storageQuota         BigInt    @default(10737418240) // 10GB default
  storageUsed          BigInt    @default(0)
  isActive             Boolean   @default(true)
  lastLoginAt          DateTime?

  // TOTP (Two-Factor Authentication) - MANDATORY for all users
  totpSecret   String? // Encrypted TOTP secret (if exists, TOTP is enabled)
  totpVerified Boolean @default(false) // Whether TOTP setup is complete

  // Token versioning - increment on password change to invalidate all tokens
  tokenVersion Int @default(0)

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  // Relations
  deviceSessions  DeviceSession[]
  totpBackupCodes TotpBackupCode[]
  auditLogs       AuditLog[]

  @@index([email])
  @@index([isActive])
  @@index([role])
  @@map("users")
}

// Device Session Model
model DeviceSession {
  id           String     @id @default(cuid())
  userId       String
  deviceType   DeviceType // MOBILE or WEB
  deviceName   String // User-friendly device name
  deviceId     String // Unique device identifier
  userAgent    String? // Browser/App user agent
  ipAddress    String? // IP address
  location     String? // Approximate location (city, country)
  accessToken  String     @unique @db.VarChar(500) // JWT tokens (typically 200-500 chars)
  refreshToken String     @unique @db.VarChar(500) // JWT tokens (typically 200-500 chars)
  expiresAt    DateTime
  lastActiveAt DateTime   @default(now())
  isActive     Boolean    @default(true)
  isRevoked    Boolean    @default(false)
  createdAt    DateTime   @default(now())
  updatedAt    DateTime   @updatedAt

  // Relations
  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@index([userId])
  @@index([deviceId])
  @@index([accessToken])
  @@index([refreshToken])
  @@index([expiresAt])
  @@index([isActive])
  @@map("device_sessions")
}

// TOTP Backup Codes Model
model TotpBackupCode {
  id        String    @id @default(cuid())
  userId    String
  code      String // Hashed backup code
  isUsed    Boolean   @default(false)
  usedAt    DateTime?
  createdAt DateTime  @default(now())

  // Relations
  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@index([userId])
  @@index([isUsed])
  @@map("totp_backup_codes")
}

// Audit Log Model - Security event logging
model AuditLog {
  id        String   @id @default(cuid())
  userId    String?  // Null for system events
  eventType String   // LOGIN, LOGOUT, LOGIN_FAILED, PASSWORD_CHANGE, TOKEN_REFRESH, etc.
  eventData Json?    // Additional event data (IP, device, etc.)
  ipAddress String?
  userAgent String?
  success   Boolean  @default(true)
  errorMessage String?
  createdAt DateTime @default(now())

  // Relations
  user User? @relation(fields: [userId], references: [id], onDelete: SetNull)

  @@index([userId])
  @@index([eventType])
  @@index([createdAt])
  @@index([success])
  @@map("audit_logs")
}
